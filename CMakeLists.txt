cmake_minimum_required(VERSION 3.16)
project(uav_predict_sim VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(yaml-cpp REQUIRED)
add_library(uav_physics src/dynamics/quadrotor.cpp src/flatness/map.cpp
  src/trajectory/reference.cpp src/control/so3.cpp)
target_include_directories(uav_physics PUBLIC include)
target_link_libraries(uav_physics PUBLIC Eigen3::Eigen)
target_compile_options(uav_physics PRIVATE -Wall -Wextra -Wpedantic)
add_library(uav_experiment src/prediction/predictor.cpp src/simulation/experiment.cpp
  src/io/config.cpp src/io/records.cpp)
target_link_libraries(uav_experiment PUBLIC uav_physics yaml-cpp)
target_compile_options(uav_experiment PRIVATE -Wall -Wextra -Wpedantic)
execute_process(COMMAND git describe --always --dirty WORKING_DIRECTORY ${PROJECT_SOURCE_DIR}
  OUTPUT_VARIABLE UAV_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT UAV_REVISION)
  set(UAV_REVISION unknown)
endif()
target_compile_definitions(uav_experiment PRIVATE UAV_REVISION="${UAV_REVISION}")
add_executable(uav_sim apps/simulate.cpp)
target_link_libraries(uav_sim PRIVATE uav_experiment)
include(CTest)
if(BUILD_TESTING)
  find_package(Boost REQUIRED)
  add_executable(physics_tests tests/physics_tests.cpp)
  target_link_libraries(physics_tests PRIVATE uav_physics)
  add_test(NAME physics COMMAND physics_tests)
  add_executable(control_tests tests/control_tests.cpp)
  target_link_libraries(control_tests PRIVATE uav_physics)
  add_test(NAME flatness_control COMMAND control_tests)
  add_executable(upstream_regression tests/upstream_regression.cpp
    tests/upstream/Quadrotor.cpp tests/upstream/SO3Control.cpp)
  target_include_directories(upstream_regression PRIVATE tests/upstream/include ${Boost_INCLUDE_DIRS})
  target_link_libraries(upstream_regression PRIVATE uav_physics)
  add_test(NAME upstream_regression COMMAND upstream_regression)
  add_executable(experiment_tests tests/experiment_tests.cpp)
  target_link_libraries(experiment_tests PRIVATE uav_experiment)
  add_test(NAME causal_experiment COMMAND experiment_tests)
endif()
