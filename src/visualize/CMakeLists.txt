cmake_minimum_required(VERSION 3.16)
project(prediction_visualize LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(Pangolin REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
add_executable(prediction_viewer src/viewer.cpp)
target_include_directories(prediction_viewer PRIVATE include)
target_link_libraries(prediction_viewer pango_display pango_plot pango_opengl pango_image pango_geometry Eigen3::Eigen)
ament_target_dependencies(prediction_viewer rclcpp nav_msgs std_msgs ament_index_cpp)
add_executable(scene_viewer src/scene_viewer.cpp)
target_link_libraries(scene_viewer pango_display pango_opengl)
ament_target_dependencies(scene_viewer rclcpp nav_msgs sensor_msgs)
add_executable(prediction_plot src/plot.cpp)
target_compile_definitions(prediction_plot PRIVATE WITHOUT_NUMPY)
target_include_directories(prediction_plot PRIVATE third_party)
target_link_libraries(prediction_plot Python3::Python)
install(TARGETS prediction_viewer scene_viewer prediction_plot DESTINATION lib/${PROJECT_NAME})
install(FILES third_party/matplotlib-cpp.LICENSE DESTINATION share/${PROJECT_NAME})
install(DIRECTORY meshes DESTINATION share/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_drone_pose test/test_drone_pose.cpp)
  target_include_directories(test_drone_pose PRIVATE include)
  target_link_libraries(test_drone_pose Eigen3::Eigen)
  ament_target_dependencies(test_drone_pose geometry_msgs)
  add_test(NAME drone_pose_transform COMMAND test_drone_pose)
endif()
ament_package()
