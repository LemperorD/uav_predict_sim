cmake_minimum_required(VERSION 3.16)
project(predictor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(quadrotor_msgs REQUIRED)
add_library(imm src/imm.cpp)
target_include_directories(imm PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(imm PUBLIC Eigen3::Eigen)
add_library(predictor_algorithms src/imm_predictor.cpp src/Pose-KF.cpp src/predictor_factory.cpp)
target_include_directories(predictor_algorithms PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(predictor_algorithms PUBLIC imm)
add_executable(predictor_node src/predictor_node.cpp)
target_link_libraries(predictor_node predictor_algorithms)
ament_target_dependencies(predictor_node rclcpp nav_msgs std_msgs)
add_executable(pose_kf_replay experiments/pose_kf_replay.cpp)
target_link_libraries(pose_kf_replay predictor_algorithms)
add_executable(imm_pose_replay experiments/imm_pose_replay.cpp)
target_link_libraries(imm_pose_replay imm)
add_executable(kinematic_source src/kinematic_source.cpp)
target_include_directories(kinematic_source PRIVATE include)
ament_target_dependencies(kinematic_source rclcpp quadrotor_msgs std_msgs)
install(TARGETS imm predictor_algorithms EXPORT immTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS predictor_node kinematic_source imm_pose_replay pose_kf_replay DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY core DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME predictor_optional_diagnostics
           COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_simulation_summary.py)
  add_executable(test_predictor_interface test/test_predictor_interface.cpp)
  target_link_libraries(test_predictor_interface predictor_algorithms)
  add_test(NAME predictor_interface COMMAND test_predictor_interface)
  add_executable(test_pose_kf test/test_pose_kf.cpp)
  target_link_libraries(test_pose_kf predictor_algorithms)
  add_test(NAME pose_kf_numerics COMMAND test_pose_kf)
  add_executable(test_imm_pose test/test_imm_pose.cpp)
  target_link_libraries(test_imm_pose imm)
  add_test(NAME imm_pose_hypothesis COMMAND test_imm_pose)
  add_executable(test_imm test/test_imm.cpp)
  target_link_libraries(test_imm imm)
  add_test(NAME imm_numerics COMMAND test_imm)
  add_executable(test_attitude test/test_attitude.cpp)
  target_link_libraries(test_attitude imm)
  add_test(NAME imm_attitude_models COMMAND test_attitude)
  add_executable(test_trajectory test/test_trajectory.cpp)
  target_include_directories(test_trajectory PRIVATE include)
  add_test(NAME trajectory_kinematics COMMAND test_trajectory)
endif()
ament_export_targets(immTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3)
ament_package()
